\section*{Selected Publications}
\paper{https://arxiv.org/abs/2606.22652}
{A Markov Chain Approach to Preference Alignment}
{\textbf{Takuya Koriyama} and Tengyuan Liang}
{Preprint, arXiv:2606.22652, 2026}
{}
{}
\recognition{\color{gray}Honorable Mention for New Solution Concept, ICML 2026 NExT-Game Workshop}
Introduces Markov Chain from Human Feedback to align generative models directly from pairwise preferences. Establishes convergence guarantees that adapt to non-transitive preferences and connects the method to reinforcement learning and game-theoretic alignment.
\entrygap


\paper{https://arxiv.org/abs/2409.15252}
{Precise Asymptotics of Bagging Regularized M-estimators}
{\textbf{Takuya Koriyama}, Pratik Patil, Jin-Hong Du, Kai Tan, and Pierre C. Bellec}
{\textbf{The Annals of Statistics}, 2026}
{https://projecteuclid.org/journals/annals-of-statistics/volume-54/issue-2/Precise-asymptotics-of-bagging-regularized-M-estimators/10.1214/25-AOS2590.short}
{Journal}
\recognition{\color{gray}Selected for the ICML Journal-to-Conference track}
Derives precise prediction-error formulas and a consistent risk estimator for ensembles trained on overlapping subsamples. Shows how jointly tuning subsample size, ensemble size, and regularization can improve predictive performance.
\entrygap

\paper{https://arxiv.org/abs/2506.21527}
{Asymptotic Inference for Exchangeable Gibbs Partition}
{\textbf{Takuya Koriyama}}
{\textbf{Stochastic Processes and their Applications}, 2026}
{https://www.sciencedirect.com/science/article/pii/S0304414926002061}
{Journal}
Develops parameter estimation and predictive uncertainty quantification for random partition models, including asymptotically valid confidence intervals and error quantification for predicting allocation to existing or new groups.
\entrygap

\paper{https://arxiv.org/abs/2411.01629}
{Denoising Diffusions with Optimal Transport: Localization, Curvature, and Multi-Scale Complexity}
{Tengyuan Liang, Kulunu Dharmakeerthi, and \textbf{Takuya Koriyama}}
{\textbf{Transactions on Machine Learning Research}, 2026}
{https://openreview.net/pdf?id=sj1wU6gBXH}
{Journal}
\recognition{\color{gray}Selected for Journal-to-Conference certification}
Characterizes score-based denoising through optimal transport and quantifies how data geometry across noise scales determines denoising difficulty. Identifies bottleneck noise levels for complex, non-log-concave distributions.
\entrygap

\paper{https://arxiv.org/abs/2310.01374}
{Corrected generalized cross-validation for finite ensembles of penalized estimators}
{Pierre C. Bellec, Jin-Hong Du, \textbf{Takuya Koriyama}, Pratik Patil, and Kai Tan (alphabetical)}
{\textbf{Journal of the Royal Statistical Society: Series B}, 87(2), 289--318, 2025}
{https://doi.org/10.1093/jrsssb/qkae092}
{Journal}
Develops corrected generalized cross-validation for estimating prediction error in ensembles of penalized least-squares models. Provides statistical guarantees for an efficient evaluation and tuning method that avoids holding out training data or repeatedly refitting models.
